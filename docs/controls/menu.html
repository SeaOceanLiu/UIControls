4.12 Menu（菜单：MenuBar / MenuPanel / MenuItem）

菜单系统由三件套组成：MenuBar（顶级菜单栏）、MenuPanel（下拉面板，支持嵌套子菜单）、MenuItem（菜单项，支持 Normal / Separator / SubMenu）。三者均继承 ControlImpl。支持「菜单模式」：点击展开后鼠标移动自动切换顶级菜单。

4.12.1 控件的创建

// C++（三件套 + Builder）
MenuItem(Control* parent, MenuItemType type = MenuItemType::Normal, float xScale = 1.0f, float yScale = 1.0f);
MenuPanel(Control* parent, float xScale = 1.0f, float yScale = 1.0f);
MenuBar(Control* parent, float xScale = 1.0f, float yScale = 1.0f);

auto filePanel = MenuPanelBuilder()
    .addItem(MenuItemBuilder(u8"新建(N)").setShortcut("Ctrl+N").setOnClick(cb).build())
    .addSeparator()
    .build();
auto bar = MenuBarBuilder(BENCH).addMenu(u8"文件(F)", filePanel).build();
BENCH->addControl(bar);

// C ABI（组装顺序：MenuItem → MenuPanel → MenuBar）
UIControlHandle item  = UICornerstone_CreateMenuItem(g_inst, "Open", 0);          // 0=Normal 1=Separator 2=SubMenu
UIControlHandle panel = UICornerstone_CreateMenuPanel(g_inst);
UICornerstone_MenuPanelAddItem(g_inst, panel, item);
UIControlHandle bar   = UICornerstone_CreateMenuBar(g_inst, 0, 0, 800, 32);
UICornerstone_MenuBarAddMenu(g_inst, bar, "File", panel);

4.12.2 控件的方法

MenuItem

方法说明
setCaption(string) / getCaption()菜单项文字
setShortcut(string) / getShortcut()快捷键文本（如 "Ctrl+N"）
setChecked(bool) / getChecked()勾选状态
setOnClick(OnClickHandler)点击回调
setSubMenu(shared_ptr<MenuPanel>) / getSubMenu() / hasSubMenu()子菜单
getType()项类型（Normal / Separator / SubMenu）
setMenuFont(SharedFont, float) / getFont() / getFontSize()面板注入共享字体
closeMenuChain()关闭整个菜单链

MenuPanel

方法说明
addItem(shared_ptr<MenuItem>) / addSeparator() / removeItem(...)菜单项管理
show() / hide() / isVisible() / setPosition(x, y)显隐与定位
getItemAt(x, y)命中测试
closeWithChildren() / recalculateSize()关闭 / 重算尺寸
getOpenSubMenu() / setOpenSubMenu(panel)、getHoveredIndex() / setHoveredIndex(int)子菜单 / 悬停状态
setFontSize(float) / getFontSize()、setItemHeightRatio(float) / getItemHeightRatio()（1.0~3.0，默认 1.6）、setFontName(FontName)字体 / 行高比 / 字体名（即时生效）

MenuBar

方法说明
addMenu(caption, panel) / removeMenu(caption)顶级菜单管理
closeAllMenus()关闭全部下拉
setBarHeight(float) / getBarHeight()菜单栏高度
setItemHeightRatio(float) / getItemHeightRatio()、setFontSize(float) / getFontSize()尺寸 / 字号（默认 20）
isInMenuMode() / enterMenuMode(int index) / exitMenuMode()菜单模式（点击展开后鼠标移动自动切换）

4.12.6 C++ Binding / 核心库样例

// 核心库（Builder 链式组装）
g_filePanel = MenuPanelBuilder()
    .addItem(MenuItemBuilder(u8"新建(N)")
        .setShortcut("Ctrl+N")
        .setOnClick([](shared_ptr item) { cout << "New file clicked" << endl; })
        .build())
    .addSeparator()
    .addItem(MenuItemBuilder(u8"最近打开的文件")
        .setSubMenu(MenuPanelBuilder()
            .addItem(MenuItemBuilder(u8"file1.txt")
                .setOnClick([](shared_ptr item) { cout << "file1 clicked" << endl; })
                .build())
            .build())
        .build())
    .build();
g_menuBar = MenuBarBuilder(BENCH)
    .addMenu(u8"文件(F)", g_filePanel)
    .build();
BENCH->addControl(g_menuBar);

// Binding：MenuItem 的 caption/checked/shortcut/click 走统一属性系统
Control item = ui->CreateMenuItem("Open", 0);
item.SetString("caption", "Open");
item.SetCallback("click", [](const Event&){ /* ... */ });
